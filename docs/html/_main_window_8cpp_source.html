468void MainWindow::onGenerateAutoPlan() {
469 try {
470 int g = goalCombo->currentIndex();
471 int durationIndex = durationCombo->currentIndex();
472 QDate d = QDate::currentDate();
473
474 int daysToPlan = 30;
475 if (durationIndex == 1) {
476 daysToPlan = 183;
477 } else if (durationIndex == 2) {
478 daysToPlan = 365;
479 }
480
481 for(int i = 0; i < daysToPlan; ++i) {
482 QDate cur = d.addDays(i);
483 if(cur.dayOfWeek() == 1 || cur.dayOfWeek() == 3 || cur.dayOfWeek() == 5) {
484 plannedDates[cur] = {true, g, {}};
485 }
486 }
487 syncPlanToDB();
488 updateCalendarColors();
489 updateHistoryUI();
490 autoPlanWidget->setVisible(false);
491 } catch(const std::exception& e) {
