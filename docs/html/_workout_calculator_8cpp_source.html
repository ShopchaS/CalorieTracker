67int WorkoutCalculator::calculateDailyCalories(const User& u) {
68 double bmr = (u.gender == 'M')
69 ? (10 * u.weight + 6.25 * u.height - 5 * u.age + 5)
70 : (10 * u.weight + 6.25 * u.height - 5 * u.age - 161);
71 return static_cast<int>(bmr * u.activity_level);
